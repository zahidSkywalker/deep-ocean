     7→const T = 16;         // tile size in pixels
     8→const S = 3;          // scale factor
     9→const ST = T * S;     // scaled tile = 48px
    10→const MW = 50;        // map width in tiles
    11→const MH = 45;        // map height in tiles
    12→const DAY_SEC = 180;  // 3 real minutes = 1 game day

    15→// ═══════════════════════════════════════════════════
    16→// SEEDED RANDOM
    17→// ═══════════════════════════════════════════════════
    18→function mkRng(seed: number) {
    19→  let s = seed;
    20→  return () => { s = (s * 16807) % 2147483647; return (s - 1) / 2147483646; };
    21→}

    24→// TILE ASSIGNMENT
    25→// Pre-assign tileset indices for visual variety
    26→// ═══════════════════════════════════════════════════
    27→// Light grass tiles (rows 0-3, cols 8-11 of tileset)
    28→const GRASS = [8, 10, 11, 20, 21, 33, 34, 35, 44, 45, 47];
    29→// Darker grass (rows 4-7)
    30→const GRASS_DARK = [56, 58, 59, 68, 69, 82, 83, 92, 93, 95];
    31→// Dirt / path tiles (rows 8-15)
    32→const DIRT = [104, 106, 107, 116, 117, 129, 130, 131, 140, 141, 143, 152, 154, 155, 164, 165, 177, 178, 179, 188, 189, 191];
    33→// Farmland / tilled soil
    34→const SOIL = [104, 106, 107, 116, 129, 130, 140, 141];
    35→// Water tiles (rows 16-19)
    36→const WATER = [192, 193, 194, 196, 197, 204, 205, 206, 208, 209, 210];
    37→// Sand tiles
    38→const SAND = [216, 217, 218, 228, 229, 230];
    39→
    40→function pick(arr: number[], rng: () => number) { return arr[(rng() * arr.length) | 0]; }

    42→// ═══════════════════════════════════════════════════
    43→// MAP GENERATION
    44→// terrain: 0=grass, 1=path, 2=farmland, 3=pond edge, 4=sand, 5=water
    45→// ═══════════════════════════════════════════════════
    46→function generateMap() {
    47→  const rng = mkRng(777);
    48→  const map: number[][] = [];
    49→  // Tile index map (which tileset tile to draw)
    50→  const tiles: number[][] = [];
    51→
    52→  for (let y = 0; y < MH; y++) {
    53→    map[y] = [];
    54→    tiles[y] = [];
    55→    for (let x = 0; x < MW; x++) {
    56→      map[y][x] = 0;
    57→      tiles[y][x] = pick(GRASS, rng);
    58→    }
    59→  }
    60→
    61→  const set = (x: number, y: number, v: number) => {
    62→    if (x >= 0 && x < MW && y >= 0 && y < MH) map[y][x] = v;
    63→  };
    64→  const setTile = (x: number, y: number, v: number) => {
    65→    if (x >= 0 && x < MW && y >= 0 && y < MH) tiles[y][x] = v;
    66→  };
    67→
    68→  // ── PATHS (3 tiles wide for a nice road feel) ──
    69→  const pathH = (x1: number, x2: number, y: number) => {
    70→    for (let x = Math.min(x1, x2); x <= Math.max(x1, x2); x++) {
    71→      set(x, y, 1); set(x, y + 1, 1);
    72→    }
    73→  };
    74→  const pathV = (y1: number, y2: number, x: number) => {
    75→    for (let y = Math.min(y1, y2); y <= Math.max(y1, y2); y++) {
    76→      set(x, y, 1); set(x + 1, y, 1);
    77→    }
    78→  };
    79→  const setPathTiles = () => {
    80→    for (let y = 0; y < MH; y++)
    81→      for (let x = 0; x < MW; x++)
    82→        if (map[y][x] === 1) tiles[y][x] = pick(DIRT, rng);
    83→  };
    84→
    85→  // Main north-south road through center
    86→  pathV(4, 40, 23);
    87→  pathV(4, 40, 24);
    88→
    89→  // East-west road
    90→  pathH(5, 45, 18);
    91→  pathH(5, 45, 19);
    92→
    93→  // Branch west to farm
    94→  pathV(18, 30, 11);
    95→  pathH(11, 18, 22);
    96→
    97→  // Branch east to animals
    98→  pathV(18, 30, 36);
    99→  pathH(24, 36, 22);
   100→
   101→  // Branch south to pond
   102→  pathV(20, 35, 10);
   103→
   104→  setPathTiles();
   105→
   106→  // ── FARM AREA (west side) ──
   107→  for (let y = 23; y <= 32; y++)
   108→    for (let x = 5; x <= 17; x++)
   109→      if (map[y][x] === 0) {
   110→        map[y][x] = 2;
   111→        tiles[y][x] = pick(SOIL, rng);
   112→      }
   113→
   114→  // ── POND (southwest, organic ellipse) ──
   115→  for (let y = 33; y <= 42; y++)
   116→    for (let x = 3; x <= 14; x++) {
   117→      const dx = (x - 8.5) / 5.5, dy = (y - 37.5) / 5;
   118→      if (dx * dx + dy * dy <= 1) {
   119→        map[y][x] = 5;
   120→        tiles[y][x] = pick(WATER, rng);
   121→      }
   122→    }
   123→  // Sandy edges
   124→  for (let y = 32; y <= 43; y++)
   125→    for (let x = 2; x <= 15; x++) {
   126→      if (map[y][x] === 5) continue;
   127→      let adj = false;
   128→      for (let dy2 = -1; dy2 <= 1; dy2++)
   129→        for (let dx2 = -1; dx2 <= 1; dx2++) {
   130→          const ny = y + dy2, nx = x + dx2;
   131→          if (ny >= 0 && ny < MH && nx >= 0 && nx < MW && map[ny][nx] === 5) adj = true;
   132→        }
   133→      if (adj && map[y][x] === 0) {
   134→        map[y][x] = 4;
   135→        tiles[y][x] = pick(SAND, rng);
   136→      }
   137→    }
   138→
   139→  // Add grass variation to remaining grass tiles
   140→  for (let y = 0; y < MH; y++)
   141→    for (let x = 0; x < MW; x++)
   142→      if (map[y][x] === 0 && rng() < 0.35)
   143→        tiles[y][x] = pick(GRASS_DARK, rng);
   144→
   145→  return { map, tiles };
   146→}

   151→interface Obj { type: string; x: number; y: number; v: number; sortY: number; }

   153→function generateObjects(): Obj[] {
   154→  const rng = mkRng(42);
   155→  const objs: Obj[] = [];
   156→  const addTree = (x: number, y: number, v: number) => {
   157→    objs.push({ type: 'tree', x, y, v: v % 5, sortY: y + 2.5 });
   158→  };
   159→
   160→  // ── Dense border trees ──
   161→  for (let x = 0; x < MW; x++) {
   162→    if (x % 2 === 0) addTree(x, 0, rng() * 5 | 0);
   163→    if (x % 3 === 0) addTree(x, 1, rng() * 5 | 0);
   164→    if (x % 3 === 0) addTree(x, MH - 2, rng() * 5 | 0);
   165→    if (x % 2 === 0) addTree(x, MH - 1, rng() * 5 | 0);
   166→  }
   167→  for (let y = 2; y < MH - 2; y++) {
   168→    if (y % 2 === 0) addTree(0, y, rng() * 5 | 0);
   169→    if (y % 3 === 0) addTree(1, y, rng() * 5 | 0);
   170→    if (y % 3 === 0) addTree(MW - 2, y, rng() * 5 | 0);
   171→    if (y % 2 === 0) addTree(MW - 1, y, rng() * 5 | 0);
   172→  }
   173→
   174→  // ── Tree clusters (forested areas) ──
   175→  const clusters: [number, number, number, number][] = [
   176→    [3, 2, 5, 3],     // NW corner
   177→    [42, 2, 6, 3],    // NE corner
   178→    [2, 40, 4, 3],    // SW corner
   179→    [40, 38, 7, 5],   // SE corner
   180→    [28, 32, 4, 3],   // SE interior
   181→  ];
   182→  for (const [cx, cy, w, h] of clusters)
   183→    for (let dy = 0; dy < h; dy++)
   184→      for (let dx = 0; dx < w; dx++)
   185→        if (rng() < 0.55) addTree(cx + dx, cy + dy, rng() * 5 | 0);
   186→
   187→  // ── Scattered individual trees ──
   188→  const treePositions = [
   189→    [7, 10], [35, 8], [44, 12], [20, 36], [30, 5], [15, 14],
   190→    [46, 22], [4, 28], [48, 30], [32, 10], [18, 8], [38, 6],
   191→    [8, 20], [42, 18], [20, 5], [35, 35],
   192→  ];
   193→  for (const [x, y] of treePositions) addTree(x, y, rng() * 5 | 0);
   194→
   195→  // ── Flower patches (near house, along paths, near pond) ──
   196→  const flowerSpots = [
   197→    // Near house
   198→    [18, 13], [19, 13], [31, 13], [32, 13],
   199→    [17, 14], [33, 14],
   200→    // Along paths
   201→    [22, 16], [27, 16], [22, 21], [27, 21],
   202→    [21, 25], [28, 25], [21, 30], [28, 30],
   203→    // Near pond
   204→    [15, 34], [16, 36], [15, 38], [2, 36], [3, 38],
   205→    // Scattered
   206→    [38, 10], [42, 14], [8, 15], [45, 25],
   207→  ];
   208→  for (const [x, y] of flowerSpots)
   209→    objs.push({ type: 'flower', x, y, v: rng() * 4 | 0, sortY: y + 0.3 });
   210→
   211→  // ── Rocks ──
   212→  const rockSpots = [[16, 12], [38, 14], [5, 30], [45, 20], [30, 8], [40, 32], [10, 16], [35, 6], [12, 38]];
   213→  for (const [x, y] of rockSpots)
   214→    objs.push({ type: 'rock', x, y, v: rng() * 3 | 0, sortY: y + 0.5 });
   215→
   216→  // ── Chest near house ──
   217→  objs.push({ type: 'chest', x: 32, y: 12, v: 0, sortY: 12.5 });
   218→
   219→  return objs;
   220→}

   381→      // ── UPDATE ──
   382→      function moveToward(e: { x: number; y: number; tx: number; ty: number; dir: number; frame: number; ft: number; moving: boolean }, spd: number, dt: number) {
   383→        if (!e.moving) return;
   384→        const dx = e.tx - e.x, dy = e.ty - e.y;
   385→        const d = Math.sqrt(dx * dx + dy * dy);
   386→        if (d < 0.15) { e.x = e.tx; e.y = e.ty; e.moving = false; return; }
   387→        const ms = spd * dt;
   388→        e.x += (dx / d) * ms;
   389→        e.y += (dy / d) * ms;
   390→        e.dir = Math.abs(dx) > Math.abs(dy) * 0.5 ? (dx > 0 ? 2 : 1) : (dy > 0 ? 0 : 3);
   391→        e.ft += dt;
   392→        if (e.ft > 0.15) { e.ft -= 0.15; e.frame = (e.frame + 1) % 4; }
   393→      }

   395→      function updateFarmer(dt: number) {
   396→        const newR = getRoutine();
   397→        if (routines.indexOf(newR) !== rIdx) { rIdx = routines.indexOf(newR); wIdx = 0; waiting = false; }

   400→          if (farmer.ft > 0.5) { farmer.ft -= 0.5; farmer.frame = (farmer.frame + 1) % 2; }
   401→          farmer.st -= dt;
   402→          if (farmer.st <= 0) { wIdx++; waiting = false; }
   403→          return;
   404→        }
   405→        const routine = routines[rIdx];
   406→        const wp = routine.wps[Math.min(wIdx, routine.wps.length - 1)];
   407→        if (!farmer.moving) { farmer.tx = wp.x; farmer.ty = wp.y; farmer.moving = true; }
   408→        moveToward(farmer, SPEED, dt);
   409→        if (!farmer.moving) {
   410→          farmer.state = wp.action || 'idle';
   411→          waiting = true;
   412→          farmer.st = wp.wait;
   413→          farmer.ft = 0;
   414→          farmer.dir = 0;
   415→          farmer.frame = 0;
   416→        }
   417→        if (farmer.moving && ptcRng() < 0.08) {
   418→          ptcs.push({ x: farmer.x + (ptcRng() - 0.5), y: farmer.y + 0.5, vx: (ptcRng() - 0.5) * 0.3, vy: -0.2 - ptcRng() * 0.2, life: 0, max: 0.4 + ptcRng() * 0.3, type: 'dust', color: '#c9a96e', sz: 1 + ptcRng(), ph: 0 });
   419→        }
   420→      }

   422→      function updateAnimal(a: Animal, dt: number) {
   423→        a.st -= dt;
   424→        if (!a.moving) {
   425→          if (a.st <= 0) {
   426→            if (ptcRng() < 0.6) {
   427→              a.tx = animalBounds.x1 + 1 + ptcRng() * (animalBounds.x2 - animalBounds.x1 - 2);
   428→              a.ty = animalBounds.y1 + 1 + ptcRng() * (animalBounds.y2 - animalBounds.y1 - 2);
   429→              a.moving = true;
   430→              a.st = 3 + ptcRng() * 5;
   431→            } else { a.st = 2 + ptcRng() * 4; }
   432→          }
   433→          a.ft += dt;
   434→          if (a.ft > 0.6) { a.ft -= 0.6; a.frame = (a.frame + 1) % 2; }
   435→        } else {
   436→          moveToward(a, a.type === 'cow' ? 0.6 : 1.2, dt);
   437→          if (!a.moving) a.st = 1 + ptcRng() * 3;
   438→        }

   441→      function updateParticles(dt: number, t: number) {
   442→        const hour = gameTime % 24;
   443→        const isDay = hour >= 6 && hour < 19;
   444→        if (isDay) {
   445→          if (ptcRng() < 0.012) ptcs.push({ x: camX / ST + ptcRng() * canvas.width / ST, y: camY / ST + ptcRng() * canvas.height / ST, vx: (ptcRng() - 0.5) * 0.6, vy: (ptcRng() - 0.5) * 0.3, life: 0, max: 8 + ptcRng() * 10, type: 'butterfly', color: ['#f9a8d4', '#fbbf24', '#a78bfa', '#34d399'][ptcRng() * 4 | 0], sz: 2 + ptcRng() * 2, ph: ptcRng() * 6.28 });
   446→          if (ptcRng() < 0.006) ptcs.push({ x: camX / ST + ptcRng() * canvas.width / ST, y: camY / ST + ptcRng() * canvas.height / ST, vx: 0.3 + ptcRng() * 0.5, vy: 0.1 + ptcRng() * 0.2, life: 0, max: 5 + ptcRng() * 8, type: 'seed', color: '#fef3c7', sz: 1 + ptcRng(), ph: ptcRng() * 6.28 });
   447→        } else {
   448→          if (ptcRng() < 0.025) ptcs.push({ x: camX / ST + ptcRng() * canvas.width / ST, y: camY / ST + ptcRng() * canvas.height / ST, vx: (ptcRng() - 0.5) * 0.2, vy: (ptcRng() - 0.5) * 0.2, life: 0, max: 3 + ptcRng() * 5, type: 'firefly', color: '#fde047', sz: 2 + ptcRng() * 2, ph: ptcRng() * 6.28 });
   449→        }
   450→        for (let i = ptcs.length - 1; i >= 0; i--) {
   451→          const p = ptcs[i];
   452→          p.life += dt;
   453→          if (p.life > p.max) { ptcs.splice(i, 1); continue; }
   454→          p.x += (p.vx + (p.type === 'butterfly' ? Math.sin(t * 2 + p.ph) * 0.3 : 0)) * dt;
   455→          p.y += (p.vy + (p.type === 'firefly' ? Math.cos(t * 2.5 + p.ph) * 0.1 : 0)) * dt;
   456→        }
   457→      }

   867→            const ang = (a / 8) * 6.28 + gt * 0.3;
   868→            ctx.beginPath();
   869→            ctx.moveTo(ix + Math.cos(ang) * 9, iy + Math.sin(ang) * 9);
   870→            ctx.lineTo(ix + Math.cos(ang) * 12, iy + Math.sin(ang) * 12);
   871→            ctx.stroke();
   872→          }
   873→        } else {
   874→          // Moon
   875→          ctx.fillStyle = '#e2e8f0';
   876→          ctx.beginPath();
   877→          ctx.arc(ix, iy, 7, 0, 6.28);
   878→          ctx.fill();
   879→          ctx.fillStyle = 'rgba(20,15,10,0.85)';
   880→          ctx.beginPath();
   881→          ctx.arc(ix + 2, iy - 1, 6, 0, 6.28);
   882→          ctx.fill();
   883→        }
   884→
   885→        // Time text
   886→        ctx.fillStyle = '#e8d5b7';
   887→        ctx.font = 'bold 15px monospace';
   888→        ctx.textAlign = 'left';
   889→        ctx.fillText(timeStr, px + 36, py + 26);
   890→
   891→        // Day + Season
   892→        ctx.fillStyle = '#a89070';
   893→        ctx.font = '12px monospace';
   894→        ctx.fillText(`Day ${dayCount}`, px + 14, py + 48);
   895→
   896→        ctx.fillStyle = '#7cba5c';
   897→        ctx.font = 'bold 12px monospace';
   898→        ctx.fillText('Spring', px + 80, py + 48);
   899→
   900→        // Activity text
   901→        const stateText = farmer.state === 'farm' ? 'Tending crops' : farmer.state === 'animals' ? 'Checking animals' : farmer.state === 'pond' ? 'Admiring pond' : farmer.state === 'sleep' ? 'Sleeping...' : farmer.moving ? 'Walking...' : 'Idle';
   902→        ctx.fillStyle = 'rgba(168,144,112,0.7)';
   903→        ctx.font = '10px monospace';
   904→        ctx.fillText(stateText, px + 14, py + 68);
   905→
   906→        // ── Mini-map ──
   907→        const mmSize = 6; // mini-map tile size
   908→        const mmW = MW * mmSize;
   909→        const mmH = MH * mmSize;
   910→        const mmX = 16, mmY = h - mmH - 16;
   911→
   912→        ctx.fillStyle = 'rgba(20,15,10,0.6)';
   913→        ctx.beginPath();
   914→        ctx.roundRect(mmX - 4, mmY - 4, mmW + 8, mmH + 8, 4);
   915→        ctx.fill();
   916→
   917→        const terrainColors: Record<number, string> = {
   918→          0: '#4a8c3f', 1: '#c49a6c', 2: '#8B6914', 4: '#d4b896', 5: '#4a90d9',
   919→        };
   920→
   921→        for (let y = 0; y < MH; y++) {
   922→          for (let x = 0; x < MW; x++) {
   923→            ctx.fillStyle = terrainColors[map[y][x]] || '#4a8c3f';
   924→            ctx.fillRect(mmX + x * mmSize, mmY + y * mmSize, mmSize, mmSize);
   925→          }
   926→        }
   927→
   928→        // Camera view rectangle
   929→        const cvx = mmX + (camX / ST) * mmSize;
   930→        const cvy = mmY + (camY / ST) * mmSize;
   931→        const cvw = (canvas.width / ST) * mmSize;
   932→        const cvh = (canvas.height / ST) * mmSize;
   933→        ctx.strokeStyle = 'rgba(255,255,255,0.6)';
   934→        ctx.lineWidth = 1;
   935→        ctx.strokeRect(cvx, cvy, cvw, cvh);
   936→
   937→        // Farmer dot on mini-map
   938→        ctx.fillStyle = '#ff4444';
   939→        ctx.fillRect(mmX + farmer.x * mmSize - 1, mmY + farmer.y * mmSize - 1, 3, 3);
   940→      }
   941→
   942→      // ── GAME LOOP ──
   943→      let last = performance.now();
   944→
   945→      function loop(ts: number) {
   946→        if (!running) return;
   947→        const dt = Math.min((ts - last) / 1000, 0.1);
   948→        last = ts;
   949→
   950→        // Update time
   951→        gameTime += (24 / DAY_SEC) * dt;
   952→        if (gameTime >= 24) {
   953→          gameTime -= 24;
   954→          dayCount++;
   955→          cropStages.clear();
   956→        }
   957→
   958→        // Grow crops slowly
   959→        for (let cy = 23; cy <= 32; cy++)
   960→          for (let cx = 5; cx <= 17; cx++) {
   961→            if (map[cy]?.[cx] !== 2) continue;
   962→            const key = `${cx},${cy}`;
   963→            let st = cropStages.get(key) || 0;
   964→            if (st < 4) {
   965→              st += dt * 0.012;
   966→              const newFloor = Math.floor(st);
   967→              if (newFloor > Math.floor(cropStages.get(key) || 0)) {
   968→                cropStages.set(key, newFloor);
   969→              }
   970→            }
   971→            cropStages.set(key, Math.min(st, 4));
   972→          }
   973→
   974→        // Update entities
   975→        updateFarmer(dt);
   976→        for (const a of animals) updateAnimal(a, dt);
   977→        updateParticles(dt, ts / 1000);
   978→
   979→        // Camera: smooth follow farmer
   980→        const tx = farmer.x * ST - canvas.width / 2 + ST / 2;
   981→        const ty = farmer.y * ST - canvas.height / 2 + ST / 2;
   982→        const lf = 1 - Math.pow(0.015, dt);
   983→        camX += (tx - camX) * lf;
   984→        camY += (ty - camY) * lf;
   985→        camX = Math.max(0, Math.min(MW * ST - canvas.width, camX));
   986→        camY = Math.max(0, Math.min(MH * ST - canvas.height, camY));
   987→
   988→        render(ts / 1000);
   989→        rafId = requestAnimationFrame(loop);
   990→      }
   991→
   992→      rafId = requestAnimationFrame(loop);
   993→      return () => { running = false; cancelAnimationFrame(rafId); window.removeEventListener('resize', resize); };
   994→    }
   995→
   996→    init();
   997→    return () => { running = false; };
   998→  }, []);
   999→
  1000→  return (
  1001→    <canvas
  1002→      ref={canvasRef}
  1003→      style={{ display: 'block', width: '100vw', height: '100vh', imageRendering: 'pixelated' }}
  1004→    />
  1005→  );
  1006→}

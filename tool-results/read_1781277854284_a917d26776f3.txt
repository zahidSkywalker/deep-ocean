   114→// SEEDED RANDOM (for consistent map generation)
   115→// ═══════════════════════════════════════════════════════════════
   116→function seededRandom(seed: number): () => number {
   117→  let s = seed;
   118→  return () => {
   119→    s = (s * 16807 + 0) % 2147483647;
   120→    return (s - 1) / 2147483646;
   121→  };
   122→}

   126→// ═══════════════════════════════════════════════════════════════
   127→// MAP GENERATION
   128→// ═══════════════════════════════════════════════════════════════
   129→function generateTilemap(): number[][] {
   130→  const map: number[][] = [];
   131→
   132→  // Fill with grass variants
   133→  for (let y = 0; y < MAP_H; y++) {
   134→    map[y] = [];
   135→    for (let x = 0; x < MAP_W; x++) {
   136→      const r = rand();
   137→      if (r < 0.35) map[y][x] = T_GRASS[0];
   138→      else if (r < 0.60) map[y][x] = T_GRASS[1];
   139→      else if (r < 0.80) map[y][x] = T_GRASS[2];
   140→      else if (r < 0.92) map[y][x] = T_GRASS[3];
   141→      else map[y][x] = T_GRASS[4];
   142→    }
   143→  }
   144→
   145→  // Scatter grass detail tiles (flowers in grass)
   146→  for (let y = 2; y < MAP_H - 2; y++) {
   147→    for (let x = 2; x < MAP_W - 2; x++) {
   148→      if (rand() < 0.06) {
   149→        map[y][x] = T_GRASS_DETAIL[Math.floor(rand() * T_GRASS_DETAIL.length)];
   150→      }
   151→    }
   152→  }
   153→
   154→  // ── PATHS ──
   155→  function setPath(x: number, y: number) {
   156→    if (x >= 0 && x < MAP_W && y >= 0 && y < MAP_H) {
   157→      const variant = T_PATH[(x + y) % T_PATH.length];
   158→      map[y][x] = variant;
   159→    }
   160→  }
   161→
   162→  // Main vertical path: from house southward
   163→  for (let y = 11; y <= 38; y++) setPath(24, y);
   164→
   165→  // West branch to farm area
   166→  for (let x = 8; x <= 24; x++) setPath(x, 18);
   167→
   168→  // East branch to animal area
   169→  for (let x = 24; x <= 38; x++) setPath(x, 18);
   170→
   171→  // Path from farm gate to farm interior
   172→  for (let y = 18; y <= 22; y++) setPath(14, y);
   173→
   174→  // Path to pond
   175→  for (let y = 18; y <= 30; y++) setPath(12, y);
   176→
   177→  // Path from house to main path
   178→  for (let y = 11; y <= 12; y++) {
   179→    for (let x = 22; x <= 26; x++) setPath(x, y);
   180→  }
   181→
   182→  // Small path to animal area
   183→  for (let y = 18; y <= 22; y++) setPath(36, y);
   184→
   185→  // ── FARMLAND ──
   186→  for (let y = 20; y <= 27; y++) {
   187→    for (let x = 8; x <= 19; x++) {
   188→      // Leave a path row
   189→      if (y === 23 && x >= 8 && x <= 19) continue;
   190→      map[y][x] = T_FARMLAND[(x + y) % T_FARMLAND.length];
   191→    }
   192→  }
   193→
   194→  // ── POND ── (elliptical, centered around 12, 34)
   195→  for (let y = 30; y <= 38; y++) {
   196→    for (let x = 6; x <= 18; x++) {
   197→      const cx = 12, cy = 34;
   198→      const dx = (x - cx) / 6;
   199→      const dy = (y - cy) / 4.5;
   200→      if (dx * dx + dy * dy <= 1) {
   201→        map[y][x] = T_WATER[(x + y * 3) % T_WATER.length];
   202→      }
   203→    }
   204→  }
   205→
   206→  // Water edges (sandy border around pond)
   207→  for (let y = 29; y <= 39; y++) {
   208→    for (let x = 5; x <= 19; x++) {
   209→      if (map[y][x] >= T_WATER[0] && map[y][x] <= T_WATER[T_WATER.length - 1]) continue;
   210→      // Check if adjacent to water
   211→      let adjWater = false;
   212→      for (let dy = -1; dy <= 1; dy++) {
   213→        for (let dx = -1; dx <= 1; dx++) {
   214→          const ny = y + dy, nx = x + dx;
   215→          if (ny >= 0 && ny < MAP_H && nx >= 0 && nx < MAP_W) {
   216→            if (map[ny][nx] >= T_WATER[0] && map[ny][nx] <= T_WATER[T_WATER.length - 1]) {
   217→              adjWater = true;
   218→            }
   219→          }
   220→        }
   221→      }
   222→      if (adjWater && map[y][x] < T_WATER[0]) {
   223→        map[y][x] = T_DIRT[2]; // sandy dirt near water
   224→      }
   225→    }
   226→  }
   227→
   228→  // ── SCATTER DECORATIVE TILES ──
   229→  // Flowers in meadow areas
   230→  for (let y = 2; y < MAP_H - 2; y++) {
   231→    for (let x = 2; x < MAP_W - 2; x++) {
   232→      if (map[y][x] >= T_GRASS[0] && map[y][x] <= T_GRASS[T_GRASS.length - 1]) {
   233→        if (rand() < 0.02) {
   234→          map[y][x] = T_FLOWER[Math.floor(rand() * T_FLOWER.length)];
   235→        }
   236→        if (rand() < 0.008) {
   237→          map[y][x] = T_ROCK[Math.floor(rand() * T_ROCK.length)];
   238→        }
   239→      }
   240→    }
   241→  }
   242→
   243→  // Tall grass patches
   244→  const tallGrassSpots = [
   245→    [3, 3], [4, 5], [6, 3], [40, 4], [42, 6], [44, 3],
   246→    [3, 40], [5, 42], [42, 40], [44, 38], [46, 41],
   247→    [35, 35], [37, 37], [30, 5], [45, 15], [4, 20],
   248→  ];
   249→  for (const [tx, ty] of tallGrassSpots) {
   250→    for (let dy = 0; dy < 2; dy++) {
   251→      for (let dx = 0; dx < 2; dx++) {
   252→        const ny = ty + dy, nx = tx + dx;
   253→        if (ny >= 0 && ny < MAP_H && nx >= 0 && nx < MAP_W) {
   254→          if (map[ny][nx] >= T_GRASS[0] && map[ny][nx] <= T_GRASS[T_GRASS.length - 1]) {
   255→            map[ny][nx] = T_TALL_GRASS[Math.floor(rand() * T_TALL_GRASS.length)];
   256→          }
   257→        }
   258→      }
   259→    }
   260→  }
   261→
   262→  return map;
   263→}

   265→// ═══════════════════════════════════════════════════════════════
   266→// OBJECT GENERATION
   267→// ═══════════════════════════════════════════════════════════════
   268→function generateObjects(): MapObject[] {
   269→  const objects: MapObject[] = [];
   270→
   271→  // ── TREES ──
   272→  // Dense border (2 tiles thick)
   273→  function addTree(x: number, y: number) {
   274→    objects.push({
   275→      type: 'tree',
   276→      x, y,
   277→      variant: Math.floor(rand() * 5),
   278→      sortY: y + 2,
   279→    });
   280→  }
   281→
   282→  // Top border
   283→  for (let x = 2; x < MAP_W - 2; x++) {
   284→    if (rand() < 0.7) addTree(x, 0);
   285→    if (rand() < 0.5) addTree(x, 1);
   286→  }
   287→  // Bottom border
   288→  for (let x = 2; x < MAP_W - 2; x++) {
   289→    if (rand() < 0.5) addTree(x, MAP_H - 2);
   290→    if (rand() < 0.7) addTree(x, MAP_H - 1);
   291→  }
   292→  // Left border
   293→  for (let y = 2; y < MAP_H - 2; y++) {
   294→    if (rand() < 0.7) addTree(0, y);
   295→    if (rand() < 0.5) addTree(1, y);
   296→  }
   297→  // Right border
   298→  for (let y = 2; y < MAP_H - 2; y++) {
   299→    if (rand() < 0.5) addTree(MAP_W - 2, y);
   300→    if (rand() < 0.7) addTree(MAP_W - 1, y);
   301→  }
   302→
   303→  // Tree clusters
   304→  const treeClusters: [number, number, number, number][] = [
   305→    [3, 2, 6, 4],      // NW grove
   306→    [42, 2, 7, 4],     // NE grove
   307→    [3, 38, 5, 5],     // SW grove
   308→    [40, 35, 8, 7],    // SE forest
   309→    [28, 30, 4, 3],    // South small grove
   310→    [42, 25, 4, 3],    // East small grove
   311→  ];
   312→
   313→  for (const [cx, cy, w, h] of treeClusters) {
   314→    for (let dy = 0; dy < h; dy++) {
   315→      for (let dx = 0; dx < w; dx++) {
   316→        if (rand() < 0.6) {
   317→          addTree(cx + dx, cy + dy);
   318→        }
   319→      }
   320→    }
   321→  }
   322→
   323→  // Scattered individual trees
   324→  const scatteredTrees = [
   325→    [8, 10], [16, 4], [35, 8], [45, 12], [6, 15],
   326→    [44, 18], [3, 28], [20, 35], [35, 30], [25, 42],
   327→    [15, 14], [38, 10], [46, 22], [4, 34], [30, 40],
   328→    [22, 3], [33, 4], [48, 30], [2, 14], [47, 10],
   329→  ];
   330→  for (const [x, y] of scatteredTrees) {
   331→    if (x >= 0 && x < MAP_W && y >= 0 && y < MAP_H) {
   332→      addTree(x, y);
   333→    }
   334→  }
   335→
   336→  // ── FENCES — Farm area (cols 7-20, rows 19-28) ──
   337→  const farmMinX = 7, farmMaxX = 20, farmMinY = 19, farmMaxY = 28;
   338→  // Top fence
   339→  for (let x = farmMinX; x <= farmMaxX; x++) {
   340→    if (x === farmMinX) objects.push({ type: 'fence_corner_tl', x, y: farmMinY, variant: 0, sortY: farmMinY });
   341→    else if (x === farmMaxX) objects.push({ type: 'fence_corner_tr', x, y: farmMinY, variant: 0, sortY: farmMinY });
   342→    else objects.push({ type: 'fence_h', x, y: farmMinY, variant: 0, sortY: farmMinY });
   343→  }
   344→  // Bottom fence (with gate at x=14)
   345→  for (let x = farmMinX; x <= farmMaxX; x++) {
   346→    if (x === farmMinX) objects.push({ type: 'fence_corner_bl', x, y: farmMaxY, variant: 0, sortY: farmMaxY });
   347→    else if (x === farmMaxX) objects.push({ type: 'fence_corner_br', x, y: farmMaxY, variant: 0, sortY: farmMaxY });
   348→    else if (x === 13 || x === 14) { /* gate opening */ }
   349→    else objects.push({ type: 'fence_h', x, y: farmMaxY, variant: 1, sortY: farmMaxY });
   350→  }
   351→  // Left fence
   352→  for (let y = farmMinY + 1; y < farmMaxY; y++) {
   353→    objects.push({ type: 'fence_v', x: farmMinX, y, variant: 0, sortY: y });
   354→  }
   355→  // Right fence
   356→  for (let y = farmMinY + 1; y < farmMaxY; y++) {
   357→    objects.push({ type: 'fence_v', x: farmMaxX, y, variant: 1, sortY: y });
   358→  }
   359→
   360→  // ── FENCES — Animal area (cols 30-43, rows 19-28) ──
   361→  const animMinX = 30, animMaxX = 43, animMinY = 19, animMaxY = 28;
   362→  for (let x = animMinX; x <= animMaxX; x++) {
   363→    if (x === animMinX) objects.push({ type: 'fence_corner_tl', x, y: animMinY, variant: 1, sortY: animMinY });
   364→    else if (x === animMaxX) objects.push({ type: 'fence_corner_tr', x, y: animMinY, variant: 1, sortY: animMinY });
   365→    else objects.push({ type: 'fence_h', x, y: animMinY, variant: 2, sortY: animMinY });
   366→  }
   367→  for (let x = animMinX; x <= animMaxX; x++) {
   368→    if (x === animMinX) objects.push({ type: 'fence_corner_bl', x, y: animMaxY, variant: 1, sortY: animMaxY });
   369→    else if (x === animMaxX) objects.push({ type: 'fence_corner_br', x, y: animMaxY, variant: 1, sortY: animMaxY });
   370→    else if (x === 35 || x === 36) { /* gate opening */ }
   371→    else objects.push({ type: 'fence_h', x, y: animMaxY, variant: 3, sortY: animMaxY });
   372→  }
   373→  for (let y = animMinY + 1; y < animMaxY; y++) {
   374→    objects.push({ type: 'fence_v', x: animMinX, y, variant: 2, sortY: y });
   375→  }
   376→  for (let y = animMinY + 1; y < animMaxY; y++) {
   377→    objects.push({ type: 'fence_v', x: animMaxX, y, variant: 3, sortY: y });
   378→  }
   379→
   380→  // ── DECORATIVE FLOWERS AND ROCKS ──
   381→  const flowerSpots = [
   382→    [21, 13], [22, 14], [27, 13], [26, 15], [20, 16],
   383→    [29, 14], [28, 16], [5, 25], [45, 10], [44, 14],
   384→    [22, 30], [25, 32], [30, 32], [38, 30], [15, 8],
   385→    [35, 12], [40, 8], [10, 10], [48, 15],
   386→  ];
   387→  for (const [x, y] of flowerSpots) {
   388→    if (x >= 0 && x < MAP_W && y >= 0 && y < MAP_H) {
   389→      objects.push({ type: 'flower_dec', x, y, variant: Math.floor(rand() * 4), sortY: y });
   390→    }
   391→  }
   392→
   393→  const rockSpots = [
   394→    [22, 3], [16, 12], [38, 14], [5, 30], [45, 20],
   395→    [30, 8], [10, 16], [40, 32], [20, 40], [35, 38],
   396→  ];
   397→  for (const [x, y] of rockSpots) {
   398→    if (x >= 0 && x < MAP_W && y >= 0 && y < MAP_H) {
   399→      objects.push({ type: 'rock_dec', x, y, variant: Math.floor(rand() * 3), sortY: y });
   400→    }
   401→  }
   402→
   403→  // Chest near house
   404→  objects.push({ type: 'chest', x: 32, y: 9, variant: 0, sortY: 9 });
   405→
   406→  return objects;
   407→}

   412→function generateCrops(): CropTile[] {
   413→  const crops: CropTile[] = [];
   414→  // Farm plots: cols 8-19, rows 20-27 (excluding path row 23)
   415→  const cropTypeRows = [0, 1, 2, 3]; // different crop types from Spring Crops
   416→
   417→  for (let y = 20; y <= 27; y++) {
   418→    if (y === 23) continue; // path row
   419→    for (let x = 8; x <= 19; x++) {
   420→      // Leave some gaps for walking paths in the farm
   421→      if (x === 14 && (y === 20 || y === 21 || y === 26 || y === 27)) continue;
   422→      if ((x === 8 || x === 19) && y % 2 === 1) continue;
   423→      crops.push({
   424→        x, y,
   425→        type: cropTypeRows[(y + x) % cropTypeRows.length],
   426→        stage: Math.floor(rand() * 3), // start at various stages
   427→        progress: rand() * 0.5,
   428→      });
   429→    }
   430→  }
   431→  return crops;
   432→}

   437→function loadImage(src: string): Promise<HTMLImageElement> {
   438→  return new Promise((resolve, reject) => {
   439→    const img = new Image();
   440→    img.onload = () => resolve(img);
   441→    img.onerror = reject;
   442→    img.src = src;
   443→  });
   444→}

   471→// DRAWING HELPERS
   472→// ═══════════════════════════════════════════════════════════════
   473→function drawTileFromSheet(
   474→  ctx: CanvasRenderingContext2D,
   475→  sheet: HTMLImageElement,
   476→  tileIndex: number,
   477→  sheetCols: number,
   478→  destX: number,
   479→  destY: number
   480→) {
   481→  const srcCol = tileIndex % sheetCols;
   482→  const srcRow = Math.floor(tileIndex / sheetCols);
   483→  ctx.drawImage(
   484→    sheet,
   485→    srcCol * TILE, srcRow * TILE, TILE, TILE,
   486→    destX, destY, SCALED, SCALED
   487→  );
   488→}

   710→      function moveEntityToward(e: Entity, dt: number) {
   711→        if (!e.moving) return;

   714→        const dy = e.targetY - e.y;

   718→          e.x = e.targetX;
   719→          e.y = e.targetY;
   720→          e.moving = false;
   721→          return;
   722→        }
   723→
   724→        const moveSpeed = e.speed * dt;
   725→        const mx = (dx / dist) * moveSpeed;
   726→        const my = (dy / dist) * moveSpeed;
   727→
   728→        e.x += mx;
   729→        e.y += my;
   730→
   731→        // Direction
   732→        if (Math.abs(dx) > Math.abs(dy) * 0.5) {
   733→          e.direction = dx > 0 ? DIR_RIGHT : DIR_LEFT;
   734→        } else {
   735→          e.direction = dy > 0 ? DIR_DOWN : DIR_UP;
   736→        }

   738→        // Animation
   739→        e.frameTimer += dt;
   740→        if (e.frameTimer > 0.15) {
   741→          e.frameTimer -= 0.15;
   742→          e.frame = (e.frame + 1) % 3;
   743→        }
   744→      }

  1353→      // ── GAME LOOP ──

  1356→      function gameLoop(timestamp: number) {
  1357→        if (!running) return;
  1358→
  1359→        const dt = Math.min((timestamp - lastTime) / 1000, 0.1); // cap delta time
  1360→        lastTime = timestamp;
  1361→
  1362→        // Update
  1363→        updateTime(dt);
  1364→        updateFarmer(dt);
  1365→        for (const c of chickens) updateAnimal(c, animalBounds, dt);
  1366→        for (const c of cows) updateAnimal(c, animalBounds, dt);
  1367→        updateCrops(dt);
  1368→        updateParticles(dt, timestamp / 1000);
  1369→        updateCamera(dt);
  1370→
  1371→        // Render
  1372→        render(timestamp / 1000);
  1373→
  1374→        animFrameId = requestAnimationFrame(gameLoop);
  1375→      }
  1376→
  1377→      // Start
  1378→      animFrameId = requestAnimationFrame(gameLoop);
  1379→
  1380→      // Cleanup
  1381→      return () => {
  1382→        running = false;
  1383→        cancelAnimationFrame(animFrameId);
  1384→        window.removeEventListener('resize', resize);
  1385→      };
  1386→    }

  1389→
  1390→    return () => { running = false; };

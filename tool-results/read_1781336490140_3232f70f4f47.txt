    20→// SEEDED RANDOM
    21→// ═══════════════════════════════════════════════════
    22→function mkRng(seed: number) {
    23→  let s = seed;
    24→  return () => { s = (s * 16807) % 2147483647; return (s - 1) / 2147483646; };
    25→}

    29→// MAP GENERATION
    30→// terrain: 0=grass, 1=path, 2=farmland, 3=pond, 4=sand, 5=water
    31→// ═══════════════════════════════════════════════════
    32→function generateMap(): number[][] {
    33→  const map: number[][] = [];
    34→  for (let y = 0; y < MH; y++) {
    35→    map[y] = [];
    36→    for (let x = 0; x < MW; x++) map[y][x] = 0; // all grass
    37→  }
    38→
    39→  // Helper
    40→  const set = (x: number, y: number, v: number) => {
    41→    if (x >= 0 && x < MW && y >= 0 && y < MH) map[y][x] = v;
    42→  };
    43→  const lineH = (x1: number, x2: number, y: number, v: number) => {
    44→    for (let x = Math.min(x1, x2); x <= Math.max(x1, x2); x++) set(x, y, v);
    45→  };
    46→  const lineV = (y1: number, y2: number, x: number, v: number) => {
    47→    for (let y = Math.min(y1, y2); y <= Math.max(y1, y2); y++) set(x, y, v);
    48→  };
    49→
    50→  // ── Main path: vertical from house (col 24) going south ──
    51→  lineV(10, 40, 24, 1);
    52→  // Widen path to 2 tiles
    53→  lineV(10, 40, 25, 1);
    54→
    55→  // ── West branch to farm ──
    56→  lineH(24, 14, 20, 1);
    57→  // ── East branch to animals ──
    58→  lineH(24, 38, 20, 1);
    59→
    60→  // ── Farm path south from branch ──
    61→  lineV(20, 28, 14, 1);
    62→
    63→  // ── Animal path south from branch ──
    64→  lineV(20, 28, 38, 1);
    65→
    66→  // ── Path to pond ──
    67→  lineV(20, 32, 12, 1);
    68→  lineH(12, 8, 32, 1);
    69→
    70→  // ── Farmland area ──
    71→  for (let y = 17; y <= 27; y++)
    72→    for (let x = 7; x <= 18; x++)
    73→      if (map[y][x] === 0) map[y][x] = 2;
    74→
    75→  // ── Pond (ellipse centered at 10, 36) ──
    76→  for (let y = 32; y <= 40; y++)
    77→    for (let x = 5; x <= 15; x++) {
    78→      const dx = (x - 10) / 5.5, dy = (y - 36) / 4.5;
    79→      if (dx * dx + dy * dy <= 1) map[y][x] = 5;
    80→    }
    81→  // Sandy edges
    82→  for (let y = 31; y <= 41; y++)
    83→    for (let x = 4; x <= 16; x++) {
    84→      if (map[y][x] === 5) continue;
    85→      let adj = false;
    86→      for (let dy = -1; dy <= 1; dy++)
    87→        for (let dx = -1; dx <= 1; dx++) {
    88→          const ny = y + dy, nx = x + dx;
    89→          if (ny >= 0 && ny < MH && nx >= 0 && nx < MW && map[ny][nx] === 5) adj = true;
    90→        }
    91→      if (adj && map[y][x] === 0) map[y][x] = 4;
    92→    }
    93→
    94→  return map;
    95→}

    99→// ═══════════════════════════════════════════════════
   100→interface Obj { type: string; x: number; y: number; v: number; sortY: number; }
   101→
   102→function generateObjects(): Obj[] {
   103→  const objs: Obj[] = [];
   104→  const addTree = (x: number, y: number, v: number) => {
   105→    objs.push({ type: 'tree', x, y, v: v % 5, sortY: y + 2 });
   106→  };
   107→
   108→  // Border trees (dense)
   109→  for (let x = 1; x < MW - 1; x++) { if (rng() < 0.7) addTree(x, 0, rng() * 5 | 0); if (rng() < 0.5) addTree(x, 1, rng() * 5 | 0); }
   110→  for (let x = 1; x < MW - 1; x++) { if (rng() < 0.5) addTree(x, MH - 2, rng() * 5 | 0); if (rng() < 0.7) addTree(x, MH - 1, rng() * 5 | 0); }
   111→  for (let y = 2; y < MH - 2; y++) { if (rng() < 0.7) addTree(0, y, rng() * 5 | 0); if (rng() < 0.5) addTree(1, y, rng() * 5 | 0); }
   112→  for (let y = 2; y < MH - 2; y++) { if (rng() < 0.5) addTree(MW - 2, y, rng() * 5 | 0); if (rng() < 0.7) addTree(MW - 1, y, rng() * 5 | 0); }
   113→
   114→  // Clusters
   115→  const clusters = [[3, 2, 5, 3], [42, 2, 6, 3], [3, 38, 4, 4], [42, 36, 6, 6], [28, 30, 3, 2]];
   116→  for (const [cx, cy, w, h] of clusters)
   117→    for (let dy = 0; dy < h; dy++)
   118→      for (let dx = 0; dx < w; dx++)
   119→        if (rng() < 0.6) addTree(cx + dx, cy + dy, rng() * 5 | 0);
   120→
   121→  // Scattered
   122→  for (const [x, y] of [[8, 10], [35, 8], [44, 12], [20, 35], [30, 5], [15, 14], [46, 22], [4, 28], [48, 30]])
   123→    addTree(x, y, rng() * 5 | 0);
   124→
   125→  // Flowers
   126→  for (const [x, y] of [[22, 13], [27, 14], [20, 16], [5, 25], [45, 10], [30, 32], [38, 8], [10, 10], [22, 30], [44, 14]])
   127→    objs.push({ type: 'flower', x, y, v: rng() * 3 | 0, sortY: y });
   128→
   129→  // Rocks
   130→  for (const [x, y] of [[16, 12], [38, 14], [5, 30], [45, 20], [30, 8], [40, 32], [10, 16]])
   131→    objs.push({ type: 'rock', x, y, v: 0, sortY: y });
   132→
   133→  // Chest
   134→  objs.push({ type: 'chest', x: 30, y: 9, v: 0, sortY: 9 });
   135→
   136→  return objs;
   137→}

   250→      // ── UPDATE ──
   251→      function moveToward(e: { x: number; y: number; tx: number; ty: number; dir: number; frame: number; ft: number; moving: boolean }, spd: number, dt: number) {
   252→        if (!e.moving) return;
   253→        const dx = e.tx - e.x, dy = e.ty - e.y;
   254→        const d = Math.sqrt(dx * dx + dy * dy);
   255→        if (d < 0.1) { e.x = e.tx; e.y = e.ty; e.moving = false; return; }
   256→        const ms = spd * dt;
   257→        e.x += (dx / d) * ms;
   258→        e.y += (dy / d) * ms;
   259→        e.dir = Math.abs(dx) > Math.abs(dy) * 0.5 ? (dx > 0 ? 2 : 1) : (dy > 0 ? 0 : 3);
   260→        e.ft += dt;
   261→        if (e.ft > 0.15) { e.ft -= 0.15; e.frame = (e.frame + 1) % 4; }
   262→      }

   264→      function updateFarmer(dt: number) {

   337→      // ── RENDER ──
   338→      function render(t: number) {
   339→        const w = canvas.width, h = canvas.height;
   340→        ctx.imageSmoothingEnabled = false;

   346→        const ey = Math.min(MH, Math.ceil((camY + h) / ST) + 1);
   347→
   348→        // ── Ground tiles ──
   349→        for (let y = sy; y < ey; y++) {
   350→          for (let x = sx; x < ex; x++) {
   351→            const px = x * ST - camX;
   352→            const py = y * ST - camY;
   353→            const terrain = map[y][x];
   354→            const v = grassVariation[y][x];
   355→
   356→            if (terrain === 0) {
   357→              // Grass
   358→              const g = Math.max(0, Math.min(255, 130 + v));
   359→              const r = Math.max(0, Math.min(255, 75 + v * 0.5));
   360→              ctx.fillStyle = `rgb(${r | 0},${g | 0},48)`;
   361→            } else if (terrain === 1) {
   362→              // Path
   363→              ctx.fillStyle = (x + y) % 2 === 0 ? '#c49a6c' : '#b8905f';
   364→            } else if (terrain === 2) {
   365→              // Farmland
   366→              ctx.fillStyle = '#8B6914';
   367→            } else if (terrain === 4) {
   368→              // Sand
   369→              ctx.fillStyle = '#d4b896';
   370→            } else if (terrain === 5) {
   371→              // Water
   372→              const wp = Math.floor(t * 1.5) % 4;
   373→              const wb = wp === 0 ? 10 : wp === 1 ? 0 : wp === 2 ? -10 : 0;
   374→              ctx.fillStyle = `rgb(${74 + wb | 0},${144 + wb | 0},${217 + wb | 0})`;
   375→            }
   376→            ctx.fillRect(px, py, ST + 1, ST + 1);
   377→
   378→            // Farmland row lines
   379→            if (terrain === 2) {
   380→              ctx.fillStyle = '#6B5210';
   381→              ctx.fillRect(px, py + ST / 2 - 1, ST + 1, 2);
   382→            }
   383→          }
   384→        }
   385→
   386→        // ── Grass overlay tiles (scatter a few tileset tiles on grass) ──
   387→        if (tileset) {
   388→          for (const go of grassOverlays) {
   389→            if (go.x < sx - 1 || go.x > ex + 1 || go.y < sy - 1 || go.y > ey + 1) continue;
   390→            if (map[go.y][go.x] !== 0) continue;
   391→            const px = go.x * ST - camX;
   392→            const py = go.y * ST - camY;
   393→            const col = go.tile % 12;
   394→            const row = (go.tile / 12) | 0;
   395→            ctx.drawImage(tileset, col * T, row * T, T, T, px, py, ST, ST);
   396→          }
   397→        }
   398→
   399→        // ── Water shimmer ──
   400→        for (let y = sy; y < ey; y++) {
   401→          for (let x = sx; x < ex; x++) {
   402→            if (map[y][x] === 5) {

   684→      function drawUI(w: number) {
   685→        const hour = gameTime % 24;
   686→        const h12 = hour % 12 || 12;
   687→        const mm = Math.floor((hour % 1) * 60);
   688→        const ampm = hour < 12 ? 'AM' : 'PM';
   689→        const timeStr = `${h12}:${mm.toString().padStart(2, '0')} ${ampm}`;
   690→
   691→        // Panel
   692→        const pw = 140, ph = 72, px = w - pw - 12, py = 12;
   693→        ctx.fillStyle = 'rgba(20,15,10,0.7)';
   694→        ctx.beginPath(); ctx.roundRect(px, py, pw, ph, 6); ctx.fill();
   695→        ctx.strokeStyle = 'rgba(180,150,100,0.4)'; ctx.lineWidth = 1; ctx.stroke();
   696→
   697→        // Sun/moon icon
   698→        const ix = px + 18, iy = py + 20;
   699→        if (hour >= 6 && hour < 19) {
   700→          ctx.fillStyle = '#fbbf24';
   701→          ctx.beginPath(); ctx.arc(ix, iy, 6, 0, 6.28); ctx.fill();
   702→          ctx.strokeStyle = '#fbbf24'; ctx.lineWidth = 1.5;
   703→          const gt = performance.now() / 1000;
   704→          for (let a = 0; a < 8; a++) {
   705→            const ang = (a / 8) * 6.28 + gt * 0.3;
   706→            ctx.beginPath();
   707→            ctx.moveTo(ix + Math.cos(ang) * 8, iy + Math.sin(ang) * 8);
   708→            ctx.lineTo(ix + Math.cos(ang) * 10, iy + Math.sin(ang) * 10);
   709→            ctx.stroke();
   710→          }
   711→        } else {
   712→          ctx.fillStyle = '#e2e8f0';
   713→          ctx.beginPath(); ctx.arc(ix, iy, 6, 0, 6.28); ctx.fill();
   714→          ctx.fillStyle = 'rgba(20,15,10,0.8)';
   715→          ctx.beginPath(); ctx.arc(ix + 2, iy - 1, 5, 0, 6.28); ctx.fill();
   716→        }
   717→
   718→        ctx.fillStyle = '#e8d5b7'; ctx.font = 'bold 14px monospace'; ctx.textAlign = 'left';
   719→        ctx.fillText(timeStr, px + 32, py + 24);
   720→        ctx.fillStyle = '#a89070'; ctx.font = '11px monospace';
   721→        ctx.fillText(`Day ${dayCount}`, px + 12, py + 44);

   726→        ctx.fillStyle = 'rgba(168,144,112,0.6)'; ctx.font = '10px monospace';

   730→      // ── GAME LOOP ──
   731→      let last = performance.now();
   732→
   733→      function loop(ts: number) {
   734→        if (!running) return;
   735→        const dt = Math.min((ts - last) / 1000, 0.1);
   736→        last = ts;
   737→
   738→        // Update time
   739→        gameTime += (24 / DAY_SEC) * dt;
   740→        if (gameTime >= 24) { gameTime -= 24; dayCount++; cropStages.clear(); }
   741→
   742→        // Grow crops slowly
   743→        for (let cy = 17; cy <= 27; cy++)
   744→          for (let cx = 7; cx <= 18; cx++) {
   745→            if (map[cy]?.[cx] !== 2) continue;
   746→            const key = `${cx},${cy}`;
   747→            let st = cropStages.get(key) || 0;
   748→            if (st < 4) { st += dt * 0.01; if (st >= (cropStages.get(key) || 0) + 1) cropStages.set(key, Math.floor(st)); }
   749→            cropStages.set(key, Math.min(st, 4));
   750→          }
   751→
   752→        // Update entities
   753→        updateFarmer(dt);
   754→        for (const a of animals) updateAnimal(a, dt);
   755→        updateParticles(dt, ts / 1000);
   756→
   757→        // Camera: follow farmer
   758→        const tx = farmer.x * ST - canvas.width / 2 + ST / 2;
   759→        const ty = farmer.y * ST - canvas.height / 2 + ST / 2;
   760→        const lf = 1 - Math.pow(0.02, dt);
   761→        camX += (tx - camX) * lf;
   762→        camY += (ty - camY) * lf;
   763→        camX = Math.max(0, Math.min(MW * ST - canvas.width, camX));
   764→        camY = Math.max(0, Math.min(MH * ST - canvas.height, camY));
   765→
   766→        render(ts / 1000);
   767→        rafId = requestAnimationFrame(loop);
   768→      }
   769→
   770→      rafId = requestAnimationFrame(loop);
   771→      return () => { running = false; cancelAnimationFrame(rafId); window.removeEventListener('resize', resize); };
   772→    }
   773→
   774→    init();
   775→    return () => { running = false; };
   776→  }, []);

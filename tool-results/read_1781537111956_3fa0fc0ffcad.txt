    44→// ─── Code Block Component ───────────────────────────────────────────────────
    45→
    46→function CodeBlock({
    47→  children,
    48→  className,
    49→}: {
    50→  children: React.ReactNode;
    51→  className?: string;
    52→}) {
    53→  const [copied, setCopied] = useState(false);
    54→  const match = /language-(\w+)/.exec(className || '');
    55→  const language = match ? match[1] : '';
    56→  const codeRef = useRef<HTMLDivElement>(null);
    57→
    58→  const handleCopy = () => {
    59→    const text = codeRef.current?.innerText || '';
    60→    navigator.clipboard.writeText(text).then(() => {
    61→      setCopied(true);
    62→      setTimeout(() => setCopied(false), 2000);
    63→    });
    64→  };
    65→
    66→  return (
    67→    <div className="group relative my-4 rounded-xl overflow-hidden border border-white/[0.08] bg-[#0d1117]">
    68→      {/* Language label and copy */}
    69→      <div className="flex items-center justify-between px-4 py-2 bg-white/[0.04] border-b border-white/[0.06]">
    70→        <span className="text-xs text-zinc-500 font-mono uppercase tracking-wider">
    71→          {language || 'code'}
    72→        </span>
    73→        <button
    74→          onClick={handleCopy}
    75→          className="flex items-center gap-1.5 text-xs text-zinc-500 hover:text-zinc-300 transition-colors px-2 py-1 rounded-md hover:bg-white/[0.06]"
    76→        >
    77→          {copied ? (
    78→            <>
    79→              <Check size={12} /> Copied
    80→            </>
    81→          ) : (
    82→            <>
    83→              <Copy size={12} /> Copy
    84→            </>
    85→          )}
    86→        </button>
    87→      </div>
    88→      <div ref={codeRef} className="overflow-x-auto p-4">
    89→        <pre className="text-sm leading-relaxed">
    90→          <code className={className}>{children}</code>
    91→        </pre>
    92→      </div>
    93→    </div>
    94→  );
    95→}

   343→// ─── Model Selector ─────────────────────────────────────────────────────────
   344→
   345→function ModelSelector({
   346→  selectedModel,
   347→  onModelChange,
   348→}: {
   349→  selectedModel: string;
   350→  onModelChange: (model: string) => void;
   351→}) {
   352→  const [isOpen, setIsOpen] = useState(false);
   353→  const dropdownRef = useRef<HTMLDivElement>(null);
   354→  const currentModel = AI_CONFIG.models.find((m) => m.id === selectedModel);
   355→
   356→  useEffect(() => {
   357→    function handleClickOutside(e: MouseEvent) {
   358→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   359→        setIsOpen(false);
   360→      }
   361→    }
   362→    document.addEventListener('mousedown', handleClickOutside);
   363→    return () => document.removeEventListener('mousedown', handleClickOutside);
   364→  }, []);
   365→
   366→  return (
   367→    <div ref={dropdownRef} className="relative">
   368→      <button
   369→        onClick={() => setIsOpen(!isOpen)}
   370→        className="flex items-center gap-2 px-3 py-2 rounded-xl bg-white/[0.04] border border-white/[0.08] hover:bg-white/[0.07] transition-all duration-200 text-sm"
   371→      >
   372→        <div className="w-2 h-2 rounded-full bg-emerald-400 shrink-0" />
   373→        <span className="hidden sm:inline text-zinc-300 max-w-[160px] truncate">
   374→          {currentModel?.name || selectedModel}
   375→        </span>
   376→        <ChevronDown
   377→          size={14}
   378→          className={`text-zinc-500 transition-transform duration-200 ${isOpen ? 'rotate-180' : ''}`}
   379→        />
   380→      </button>
   381→
   382→      <AnimatePresence>
   383→        {isOpen && (
   384→          <motion.div
   385→            initial={{ opacity: 0, y: -4, scale: 0.98 }}
   386→            animate={{ opacity: 1, y: 0, scale: 1 }}
   387→            exit={{ opacity: 0, y: -4, scale: 0.98 }}
   388→            transition={{ duration: 0.15 }}
   389→            className="absolute right-0 top-full mt-2 w-72 rounded-xl bg-[#1a1d27] border border-white/[0.1] shadow-2xl shadow-black/40 z-50 overflow-hidden"
   390→          >
   391→            <div className="p-2">
   392→              <div className="px-3 py-2 text-xs font-semibold text-zinc-500 uppercase tracking-wider">
   393→                Select Model
   394→              </div>
   395→              {AI_CONFIG.models.map((model) => (
   396→                <button
   397→                  key={model.id}
   398→                  onClick={() => {
   399→                    onModelChange(model.id);
   400→                    setIsOpen(false);
   401→                  }}
   402→                  className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-left transition-colors ${
   403→                    selectedModel === model.id
   404→                      ? 'bg-purple-500/10 text-purple-300'
   405→                      : 'hover:bg-white/[0.04] text-zinc-300'
   406→                  }`}
   407→                >
   408→                  <div
   409→                    className={`w-2 h-2 rounded-full shrink-0 ${
   410→                      selectedModel === model.id ? 'bg-purple-400' : 'bg-zinc-600'
   411→                    }`}
   412→                  />
   413→                  <div className="min-w-0">
   414→                    <div className="text-sm font-medium truncate">{model.name}</div>
   415→                    <div className="text-xs text-zinc-500 truncate">{model.description}</div>
   416→                  </div>
   417→                </button>
   418→              ))}
   419→            </div>
   420→          </motion.div>
   421→        )}
   422→      </AnimatePresence>
   423→    </div>
   424→  );
   425→}

   552→  const chatContainerRef = useRef<HTMLDivElement>(null);
   553→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   554→  const abortControllerRef = useRef<AbortController | null>(null);
   555→
   556→  // Auto-scroll to bottom
   557→  const scrollToBottom = useCallback(() => {
   558→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   559→  }, []);
   560→
   561→  useEffect(() => {
   562→    if (messages.length > 0) {
   563→      scrollToBottom();
   564→    }
   565→  }, [messages, scrollToBottom]);
   566→
   567→  // Track scroll position for "scroll to bottom" button
   568→  useEffect(() => {
   569→    const container = chatContainerRef.current;
   570→    if (!container) return;
   571→
   572→    const handleScroll = () => {
   573→      const { scrollTop, scrollHeight, clientHeight } = container;
   574→      const isNearBottom = scrollHeight - scrollTop - clientHeight < 100;
   575→      setShowScrollBtn(!isNearBottom);
   576→    };
   577→
   578→    container.addEventListener('scroll', handleScroll);
   579→    return () => container.removeEventListener('scroll', handleScroll);
   580→  }, []);
   581→
   582→  // Auto-resize textarea
   583→  useEffect(() => {
   584→    const textarea = textareaRef.current;
   585→    if (!textarea) return;
   586→    textarea.style.height = 'auto';
   587→    textarea.style.height = `${Math.min(textarea.scrollHeight, 200)}px`;
   588→  }, [inputValue]);
   589→
   590→  // ─── Stream Response ─────────────────────────────────────────────────
   591→
   592→  const sendMessage = useCallback(
   593→    async (text: string) => {
   594→      if (!text.trim() || isLoading) return;
   595→
   596→      const userMessage: Message = {
   597→        id: generateId(),
   598→        role: 'user',
   599→        content: text.trim(),
   600→      };
   601→
   602→      const assistantMessage: Message = {
   603→        id: generateId(),
   604→        role: 'assistant',
   605→        content: '',
   606→      };
   607→
   608→      setMessages((prev) => [...prev, userMessage, assistantMessage]);
   609→      setInputValue('');
   610→      setIsLoading(true);
   611→      setError(null);
   612→
   613→      // Reset textarea height
   614→      if (textareaRef.current) {
   615→        textareaRef.current.style.height = 'auto';
   616→      }
   617→
   618→      const controller = new AbortController();
   619→      abortControllerRef.current = controller;
   620→
   621→      const apiMessages = [
   622→        ...messages.map((m) => ({ role: m.role, content: m.content })),
   623→        { role: userMessage.role, content: userMessage.content },
   624→      ];
   625→
   626→      try {
   627→        const response = await fetch(`${baseUrl}/chat/completions`, {
   628→          method: 'POST',
   629→          headers: {
   630→            'Content-Type': 'application/json',
   631→            Authorization: `Bearer ${apiKey}`,
   632→          },
   633→          body: JSON.stringify({
   634→            model: selectedModel,
   635→            messages: apiMessages,
   636→            stream: true,
   637→          }),
   638→          signal: controller.signal,
   639→        });
   640→
   641→        if (!response.ok) {
   642→          const errBody = await response.text().catch(() => 'Unknown error');
   643→          throw new Error(
   644→            `API returned ${response.status}: ${errBody}`
   645→          );
   646→        }
   647→
   648→        const reader = response.body?.getReader();
   649→        if (!reader) throw new Error('Failed to get response stream');
   650→
   651→        const decoder = new TextDecoder();
   652→        let fullContent = '';
   653→        let buffer = '';
   654→
   655→        while (true) {
   656→          const { done, value } = await reader.read();
   657→          if (done) break;
   658→
   659→          buffer += decoder.decode(value, { stream: true });
   660→          const lines = buffer.split('\n');
   661→          buffer = lines.pop() || ''; // Keep incomplete line in buffer
   662→
   663→          for (const line of lines) {
   664→            const trimmed = line.trim();
   665→            if (!trimmed || !trimmed.startsWith('data: ')) continue;
   666→
   667→            const data = trimmed.slice(6);
   668→            if (data === '[DONE]') continue;
   669→
   670→            try {
   671→              const parsed = JSON.parse(data);
   672→              const delta = parsed.choices?.[0]?.delta?.content;
   673→              if (delta) {
   674→                fullContent += delta;
   675→                setMessages((prev) =>
   676→                  prev.map((m) =>
   677→                    m.id === assistantMessage.id
   678→                      ? { ...m, content: fullContent }
   679→                      : m
   680→                  )
   681→                );
   682→              }
   683→            } catch {
   684→              // Skip malformed JSON
   685→            }
   686→          }
   687→        }
   688→      } catch (err: unknown) {
   689→        if (err instanceof Error && err.name === 'AbortError') {
   690→          // User stopped generation — keep partial content
   691→        } else {
   692→          const errorMsg =
   693→            err instanceof Error ? err.message : 'An unexpected error occurred';
   694→          setError(errorMsg);
   695→          // Remove empty assistant message
   696→          setMessages((prev) =>
   697→            prev.filter((m) => m.id !== assistantMessage.id)
   698→          );
   699→        }
   700→      } finally {
   701→        setIsLoading(false);
   702→        abortControllerRef.current = null;
   703→      }
   704→    },
   705→    [messages, isLoading, selectedModel, apiKey, baseUrl]
   706→  );
   707→
   708→  const stopGenerating = () => {
   709→    abortControllerRef.current?.abort();
   710→  };
   711→
   712→  const handleFormSubmit = (e: FormEvent) => {
   713→    e.preventDefault();
   714→    sendMessage(inputValue);
   715→  };
   716→
   717→  const handleKeyDown = (e: KeyboardEvent<HTMLTextAreaElement>) => {
   718→    if (e.key === 'Enter' && !e.shiftKey) {
   719→      e.preventDefault();
   720→      sendMessage(inputValue);
   721→    }
   722→  };
   723→
   724→  const clearChat = () => {
   725→    setMessages([]);
   726→    setError(null);
   727→    abortControllerRef.current?.abort();
   728→    setIsLoading(false);
   729→  };
   730→
   731→  const handleSaveSettings = (newKey: string, newUrl: string) => {
   732→    setApiKey(newKey);
   733→    setBaseUrl(newUrl);
   734→  };
   735→
   736→  return (
   737→    <div className="h-[100dvh] flex flex-col bg-[#0b0d14]">
   738→      {/* ─── Header ───────────────────────────────────────────────────── */}
   739→      <header className="shrink-0 border-b border-white/[0.06] bg-[#0b0d14]/80 backdrop-blur-xl z-10">
   740→        <div className="max-w-4xl mx-auto px-4 sm:px-6 h-16 flex items-center justify-between">
   741→          {/* Left: Logo */}
   742→          <div className="flex items-center gap-2.5">
   743→            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-purple-500 to-blue-500 flex items-center justify-center shadow-md shadow-purple-500/20">
   744→              <Sparkles size={16} className="text-white" />
   745→            </div>
   746→            <h1 className="text-lg font-bold text-white hidden sm:block">
   747→              Free AI Chat
   748→            </h1>
   749→          </div>
   750→
   751→          {/* Right: Controls */}
   752→          <div className="flex items-center gap-2">
   753→            {messages.length > 0 && !isLoading && (
   754→              <button
   755→                onClick={clearChat}
   756→                className="flex items-center gap-1.5 px-3 py-2 rounded-xl bg-white/[0.04] border border-white/[0.08] hover:bg-white/[0.07] transition-all text-sm text-zinc-400 hover:text-zinc-200"
   757→                title="New chat"
   758→              >
   759→                <RotateCcw size={14} />
   760→                <span className="hidden sm:inline">New chat</span>
   761→              </button>
   762→            )}
   763→            <ModelSelector
   764→              selectedModel={selectedModel}
   765→              onModelChange={setSelectedModel}
   766→            />
   767→            <button
   768→              onClick={() => setIsSettingsOpen(true)}
   769→              className="flex items-center justify-center w-9 h-9 rounded-xl bg-white/[0.04] border border-white/[0.08] hover:bg-white/[0.07] transition-all text-zinc-400 hover:text-zinc-200"
   770→              title="Settings"
   771→            >
   772→              <Settings2 size={16} />
   773→            </button>
   774→          </div>
   775→        </div>
   776→      </header>
   777→
   778→      {/* ─── Chat Area ───────────────────────────────────────────────── */}
   779→      <main
   780→        ref={chatContainerRef}
   781→        className="flex-1 overflow-y-auto relative"
   782→      >
   783→        <div className="max-w-4xl mx-auto px-4 sm:px-6 py-6">
   784→          {messages.length === 0 ? (
   785→            <WelcomeScreen onSendPrompt={sendMessage} />
   786→          ) : (
   787→            <div className="space-y-6 pb-4">
   788→              <AnimatePresence>
   789→                {messages.map((message) => (
   790→                  <MessageBubble
   791→                    key={message.id}
   792→                    message={message}
   793→                    isStreaming={
   794→                      isLoading &&
   795→                      message.role === 'assistant' &&
   796→                      message === messages[messages.length - 1]
   797→                    }
   798→                  />
   799→                ))}
   800→              </AnimatePresence>
   801→
   802→              {/* Error */}
   803→              {error && (
   804→                <motion.div
   805→                  initial={{ opacity: 0, y: 8 }}
   806→                  animate={{ opacity: 1, y: 0 }}
   807→                  className="flex items-start gap-2 p-4 rounded-xl bg-red-500/10 border border-red-500/20 text-red-300 text-sm"
   808→                >
   809→                  <AlertCircle size={16} className="shrink-0 mt-0.5" />
   810→                  <div>
   811→                    <p className="font-medium">Error</p>
   812→                    <p className="text-red-400/80 mt-0.5">{error}</p>
   813→                  </div>
   814→                </motion.div>
   815→              )}
   816→
   817→              <div ref={chatEndRef} />
   818→            </div>
   819→          )}
   820→        </div>
   821→
   822→        {/* Scroll to bottom button */}
   823→        <AnimatePresence>
   824→          {showScrollBtn && messages.length > 0 && (
   825→            <motion.button
   826→              initial={{ opacity: 0, scale: 0.8 }}
   827→              animate={{ opacity: 1, scale: 1 }}
   828→              exit={{ opacity: 0, scale: 0.8 }}
   829→              onClick={scrollToBottom}
   830→              className="absolute bottom-4 left-1/2 -translate-x-1/2 p-2 rounded-full bg-[#1a1d27] border border-white/[0.1] text-zinc-400 hover:text-white shadow-lg hover:bg-white/[0.06] transition-all"
   831→            >
   832→              <ArrowDown size={16} />
   833→            </motion.button>
   834→          )}
   835→        </AnimatePresence>
   836→      </main>
   837→
   838→      {/* ─── Input Area ────────────────────────────────────────────────── */}
   839→      <footer className="shrink-0 border-t border-white/[0.06] bg-[#0b0d14]/80 backdrop-blur-xl">
   840→        <div className="max-w-4xl mx-auto px-4 sm:px-6 py-4">
   841→          <form onSubmit={handleFormSubmit} className="relative">
   842→            <div className="flex items-end gap-3 p-2 rounded-2xl bg-white/[0.03] border border-white/[0.08] focus-within:border-purple-500/30 focus-within:ring-2 focus-within:ring-purple-500/10 transition-all">
   843→              <textarea
   844→                ref={textareaRef}
   845→                value={inputValue}
   846→                onChange={(e) => setInputValue(e.target.value)}
   847→                onKeyDown={handleKeyDown}
   848→                placeholder="Send a message..."
   849→                rows={1}
   850→                className="flex-1 resize-none bg-transparent text-white placeholder-zinc-600 focus:outline-none py-2.5 px-3 text-[0.935rem] leading-relaxed max-h-[200px] scrollbar-thin"
   851→                disabled={isLoading}
   852→              />
   853→              {isLoading ? (
   854→                <button
   855→                  type="button"
   856→                  onClick={stopGenerating}
   857→                  className="shrink-0 flex items-center justify-center w-10 h-10 rounded-xl bg-red-500/80 hover:bg-red-500 text-white transition-all shadow-md shadow-red-500/20"
   858→                  title="Stop generating"
   859→                >
   860→                  <Square size={14} fill="currentColor" />
   861→                </button>
   862→              ) : (
   863→                <button
   864→                  type="submit"
   865→                  disabled={!inputValue.trim()}
   866→                  className="shrink-0 flex items-center justify-center w-10 h-10 rounded-xl bg-gradient-to-r from-purple-600 to-blue-600 hover:from-purple-500 hover:to-blue-500 text-white transition-all shadow-md shadow-purple-500/20 disabled:opacity-30 disabled:cursor-not-allowed disabled:shadow-none"
   867→                  title="Send message"
   868→                >
   869→                  <Send size={16} />
   870→                </button>
   871→              )}
   872→            </div>
   873→          </form>
   874→
   875→          {/* Powered by */}
   876→          <p className="text-center text-[11px] text-zinc-600 mt-3">
   877→            Powered by{' '}
   878→            <a
   879→              href="https://routeway.ai"
   880→              target="_blank"
   881→              rel="noopener noreferrer"
   882→              className="text-zinc-500 hover:text-purple-400 transition-colors"
   883→            >
   884→              RouteWay AI
   885→            </a>{' '}
   886→            · Free models may have rate limits · Responses are not stored
   887→          </p>
   888→        </div>
   889→      </footer>
   890→
   891→      {/* ─── Settings Modal ────────────────────────────────────────────── */}
   892→      <SettingsModal
   893→        isOpen={isSettingsOpen}
   894→        onClose={() => setIsSettingsOpen(false)}
   895→        apiKey={apiKey}
   896→        baseUrl={baseUrl}
   897→        onSave={handleSaveSettings}
   898→      />
   899→    </div>
   900→  );
   901→}
